```python
"""Phase 31 H8 — real-OpenRouter money-path e2e (SPEC AC18 + AC19).

Auth → deploy nanobot recipe → send a chat → poll usage_logs for the
cost-capture row written by the proxy. Hits real OpenRouter; spends
real money (~$0.0004 per run with nanobot + openai/gpt-4o-mini per
Phase 29 baseline).

Marker: e2e_money_path (registered in pyproject.toml by Wave 0).
Run via `make e2e-money-path` which guards on $OPENROUTER_API_KEY.
"""
from __future__ import annotations

import asyncio
import os

import pytest

pytestmark = [pytest.mark.e2e_money_path, pytest.mark.asyncio]


@pytest.fixture(autouse=True)
def _require_openrouter_key():
    """Skip the test if OPENROUTER_API_KEY isn't set (defense-in-depth on top of the Makefile guard)."""
    if not os.environ.get("OPENROUTER_API_KEY"):
        pytest.skip("OPENROUTER_API_KEY not set; e2e money-path is opt-in")


async def test_chat_through_proxy_writes_usage_log(
    e2e_money_path_client, db_pool
):
    """Deploy nanobot, send a chat, assert a usage_logs row is written
    with cost_usd > 0 AND non-null upstream_request_id within 10s."""
    client = e2e_money_path_client["client"]
    cookie = e2e_money_path_client["cookie"]
    user_id = e2e_money_path_client["user_id"]
    headers = {"Cookie": cookie}

    # AMD-04: nanobot + openai/gpt-4o-mini (Phase 29 baseline ~$0.0004/run).
    # nano-kaiku does not exist in recipes/.
    deploy_resp = await client.post(
        "/v1/runs",
        headers=headers,
        json={
            "recipe_name": "nanobot",
            "model": "openai/gpt-4o-mini",
            "prompt": "Reply with one word: hello",
        },
    )
    assert deploy_resp.status_code in (200, 201, 202), (
        f"deploy failed: {deploy_resp.status_code} {deploy_resp.text}"
    )
    # The exact response shape (run_id vs agent_id vs agent_instance_id) is
    # api_server-version-dependent — adapt to the actual contract here.
    body = deploy_resp.json()
    agent_id = body.get("agent_id") or body.get("agent_instance_id") or body.get("run_id") or body.get("id")
    assert agent_id, f"deploy response missing id: {body}"

    # Send a chat. The endpoint is POST /v1/agents/{id}/messages with the
    # current message contract (text + idempotency-key headers, etc.).
    chat_resp = await client.post(
        f"/v1/agents/{agent_id}/messages",
        headers=headers,
        json={"text": "hello"},
    )
    assert chat_resp.status_code in (200, 202), (
        f"chat failed: {chat_resp.status_code} {chat_resp.text}"
    )

    # D-20: poll usage_logs for up to 10s (50 × 200ms). AMD-05 ensures
    # `usage_logs` was truncated by the autouse fixture, so this query
    # cannot match a stale row.
    usage_row = None
    for _ in range(50):
        async with db_pool.acquire() as conn:
            usage_row = await conn.fetchrow(
                "SELECT cost_usd, upstream_request_id FROM usage_logs "
                "WHERE user_id = $1 ORDER BY created_at DESC LIMIT 1",
                user_id,
            )
            if usage_row is not None:
                break
        await asyncio.sleep(0.2)

    assert usage_row is not None, (
        "usage_logs row never appeared in 10s — proxy did not capture cost; "
        "check OPENROUTER_API_KEY validity and proxy /api/v1/generation parser"
    )
    assert float(usage_row["cost_usd"]) > 0, (
        f"cost_usd should be > 0 after a real OpenRouter completion; got {usage_row['cost_usd']}"
    )
    assert usage_row["upstream_request_id"] is not None, (
        f"upstream_request_id should be the OpenRouter `gen-*` id; got {usage_row['upstream_request_id']}"
    )
```
