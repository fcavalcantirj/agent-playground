    ```dart
    // Phase 25 Wave 3 D-56 / D-57 / D-58 — multi-channel deploy orchestrator.
    // Mirror of frontend/components/playground-form.tsx lines 316-360.
    import 'package:agent_playground/core/api/api_client.dart';
    import 'package:agent_playground/core/api/dtos.dart';
    import 'package:agent_playground/core/api/result.dart';
    import 'package:dio/dio.dart';

    sealed class DeployOutcome {
      const DeployOutcome();
      const factory DeployOutcome.success({required String agentInstanceId}) = _Success;
      const factory DeployOutcome.partialSuccess({
        required String agentInstanceId,
        required String telegramFailReason,
        required Map<String, String> telegramInputs,
      }) = _PartialSuccess;
      const factory DeployOutcome.smokeFail({required String reason}) = _SmokeFail;
      const factory DeployOutcome.runsError({required ApiError error}) = _RunsError;
      const factory DeployOutcome.inappFail({required ApiError error}) = _InappFail;
      const factory DeployOutcome.cancelled() = _Cancelled;
    }
    class _Success extends DeployOutcome { const _Success({required this.agentInstanceId}); final String agentInstanceId; }
    class _PartialSuccess extends DeployOutcome { const _PartialSuccess({required this.agentInstanceId, required this.telegramFailReason, required this.telegramInputs}); final String agentInstanceId; final String telegramFailReason; final Map<String, String> telegramInputs; }
    class _SmokeFail extends DeployOutcome { const _SmokeFail({required this.reason}); final String reason; }
    class _RunsError extends DeployOutcome { const _RunsError({required this.error}); final ApiError error; }
    class _InappFail extends DeployOutcome { const _InappFail({required this.error}); final ApiError error; }
    class _Cancelled extends DeployOutcome { const _Cancelled(); }

    class DeployOrchestrator {
      DeployOrchestrator(this._api);
      final ApiClient _api;

      Future<DeployOutcome> deploy({
        required String recipeName,
        required String modelId,
        required String agentName,
        required String byokKey,
        required bool telegramEnabled,
        Map<String, String> telegramInputs = const {},
        CancelToken? cancelToken,
      }) async {
        // 1. POST /v1/runs (smoke gate per Phase 23 D-22).
        final runsRes = await _api.runs(
          body: RunRequest(recipeName: recipeName, model: modelId, agentName: agentName),
          byokOpenRouterKey: byokKey,
          cancelToken: cancelToken,
        );
        if (cancelToken?.isCancelled ?? false) return const DeployOutcome.cancelled();
        switch (runsRes) {
          case Err(:final error):
            return DeployOutcome.runsError(error: error);
          case Ok(:final value):
            if (!value.smokeOk) {
              return DeployOutcome.smokeFail(reason: 'verdict != PASS');
              // (deploy_step renders verdict.detail from a richer error
              // surface if the backend exposes it; D-30 — for now, the
              // generic reason is acceptable and tested by Test 3.)
            }
            final agentId = value.agentInstanceId;

            // 2. POST /start with channel='inapp' (D-56 always).
            final inappRes = await _api.start(
              agentId: agentId,
              byokOpenRouterKey: byokKey,
              cancelToken: cancelToken,
            );
            if (cancelToken?.isCancelled ?? false) return const DeployOutcome.cancelled();
            if (inappRes is Err<StartResponse>) {
              return DeployOutcome.inappFail(error: inappRes.error);
            }

            // 3. Optional Telegram (D-58 partial success path).
            if (telegramEnabled) {
              final tgRes = await _api.start(
                agentId: agentId,
                body: StartRequest(channel: 'telegram', channelInputs: telegramInputs),
                byokOpenRouterKey: byokKey,
                cancelToken: cancelToken,
              );
              if (cancelToken?.isCancelled ?? false) return const DeployOutcome.cancelled();
              if (tgRes is Err<StartResponse>) {
                return DeployOutcome.partialSuccess(
                  agentInstanceId: agentId,
                  telegramFailReason: tgRes.error.message,
                  telegramInputs: telegramInputs,
                );
              }
            }
            return DeployOutcome.success(agentInstanceId: agentId);
        }
      }
    }
    ```

    ```dart
    // Phase 25 Wave 3 D-54 — dynamic Telegram-input fields.
    // Mirror of frontend/components/playground-form.tsx lines 638-689.
    // Labels = recipe.channels.<id>.required_user_input[i].env verbatim.
    // NEVER hardcode 'Bot Token' / 'User ID' (Golden Rule #2).
    import 'package:agent_playground/core/api/dtos.dart';
    import 'package:agent_playground/core/theme/solvr_theme.dart';
    import 'package:agent_playground/features/new_agent/wizard_providers.dart';

    class ChannelInputs extends ConsumerWidget {
      const ChannelInputs({required this.channelMeta, super.key});
      final RecipeChannelMeta channelMeta;

      Future<void> _openExternal(String url) async {
        final uri = Uri.tryParse(url);
        if (uri == null) return;
        final scheme = uri.scheme.toLowerCase();
        if (scheme != 'https' && scheme != 'http') return;
        if (await canLaunchUrl(uri)) {
          await launchUrl(uri, mode: LaunchMode.externalApplication);
        }
      }

      @override
      Widget build(BuildContext context, WidgetRef ref) {
        final inputs = channelMeta.allInputs;
        final state = ref.watch(wizardScopeProvider);
        return Column(
          crossAxisAlignment: CrossAxisAlignment.stretch,
          children: [
            for (final input in inputs)
              Padding(
                padding: const EdgeInsets.only(top: 16),
                child: Column(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    Text(
                      input.env,
                      style: SolvrTextStyles.mono(fontSize: 14, fontWeight: FontWeight.w600),
                    ),
                    const SizedBox(height: 8),
                    TextField(
                      key: ValueKey('channel_input_${input.env}'),
                      obscureText: input.secret,
                      autocorrect: false,
                      enableSuggestions: false,
                      decoration: InputDecoration(
                        hintText: input.secret ? '••••••••' : '...',
                      ),
                      onChanged: (v) => ref
                          .read(wizardScopeProvider.notifier)
                          .setTelegramInput(input.env, v),
                    ),
                    if (input.hint != null && input.hint!.isNotEmpty)
                      Padding(
                        padding: const EdgeInsets.only(top: 4),
                        child: _HintCaption(hint: input.hint!, hintUrl: input.hintUrl, onOpen: _openExternal),
                      ),
                  ],
                ),
              ),
          ],
        );
      }
    }

    class _HintCaption extends StatelessWidget {
      const _HintCaption({required this.hint, this.hintUrl, required this.onOpen});
      final String hint;
      final String? hintUrl;
      final Future<void> Function(String) onOpen;
      @override
      Widget build(BuildContext context) {
        return Wrap(
          children: [
            Text(hint,
              style: Theme.of(context).textTheme.bodySmall?.copyWith(
                color: SolvrColors.mutedForeground,
              ),
            ),
            if (hintUrl != null && hintUrl!.isNotEmpty)
              TextButton(
                onPressed: () => onOpen(hintUrl!),
                child: const Text('get one here'),
              ),
          ],
        );
      }
    }
    ```

    ```dart
      const DeployStep({super.key});
      @override
      ConsumerState<DeployStep> createState() => _DeployStepState();

    class _DeployStepState extends ConsumerState<DeployStep> {
      final _nameCtl = TextEditingController();
      final _nameFocus = FocusNode();
      static final _kAgentNameRegex = RegExp(r'^[a-z0-9][a-z0-9_-]*$');
      static const _kAgentNameMaxLength = 64;

      // Deploy state machine.
      _DeployStateKind _state = _DeployStateKind.editing;
      String? _errorTitle;
      String? _errorBody;
      CancelToken? _cancel;
      Stopwatch? _elapsed;
      Timer? _tick;

      String? _nameValidationError(String v) {
        final trimmed = v.trim();
        if (trimmed.isEmpty) return null;
        if (trimmed.length > _kAgentNameMaxLength) return 'Name too long (max 64).';
        if (!_kAgentNameRegex.hasMatch(trimmed)) return 'Lowercase letters, numbers, _ and - only';
        return null;
      }

      Future<void> _onDeploy() async {
        final scope = ref.read(wizardScopeProvider);
        final name = scope.agentName.trim();
        if (_nameValidationError(name) != null || scope.selectedRecipe == null
            || scope.selectedModel == null || scope.byokKey.isEmpty) return;

        // D-28 — pre-flight collision check.
        final agentsRes = await ref.read(apiClientProvider).agentsList();
        if (agentsRes case Ok(:final value)) {
          final existing = value.firstWhereOrNull((a) => a.name == name);
          if (existing != null) {
            final r = await ConfirmDialog.show(
              context,
              title: 'Name "$name" already used',
              body: 'Existing agent: ${existing.recipeName} + ${existing.model}. '
                    'Re-deploy replaces it; rename keeps both.',
              cancelLabel: 'Cancel',
              confirmLabel: 'Re-deploy',
              destructive: true,
              thirdButtonLabel: 'Rename',
            );
            if (!mounted) return;
            switch (r) {
              case ConfirmDialogResult.cancel: return;
              case ConfirmDialogResult.third: _nameFocus.requestFocus(); return;
              case ConfirmDialogResult.confirm: break;
            }
          }
        }

        // D-29 — smoke loading UX.
        setState(() {
          _state = _DeployStateKind.smokeLoading;
          _elapsed = Stopwatch()..start();
        });
        _tick = Timer.periodic(const Duration(seconds: 1), (_) => setState(() {}));
        _cancel = CancelToken();

        final orch = DeployOrchestrator(ref.read(apiClientProvider));
        final outcome = await orch.deploy(
          recipeName: scope.selectedRecipe!.name,
          modelId: scope.selectedModel!.id,
          agentName: name,
          byokKey: scope.byokKey,
          telegramEnabled: scope.telegramEnabled,
          telegramInputs: scope.telegramInputs,
          cancelToken: _cancel,
        );
        _tick?.cancel();
        if (!mounted) return;

        switch (outcome) {
          case _Success(:final agentInstanceId):
            ref.read(telegramFailedBannerProvider.notifier).state = null;
            ref.read(wizardScopeProvider.notifier).clear();
            context.go('/chat/$agentInstanceId');   // D-60 replace
            return;
          case _PartialSuccess(:final agentInstanceId, :final telegramFailReason, :final telegramInputs):
            ref.read(telegramFailedBannerProvider.notifier).state = TelegramFailedBannerState(
              agentInstanceId: agentInstanceId,
              reason: telegramFailReason,
              telegramInputs: telegramInputs,
            );
            ref.read(wizardScopeProvider.notifier).clear();
            context.go('/chat/$agentInstanceId');   // D-58
            return;
          case _SmokeFail(:final reason):
            setState(() {
              _state = _DeployStateKind.smokeFail;
              _errorTitle = 'Smoke test failed';
              _errorBody = reason;
            });
            return;
          case _RunsError(:final error):
            setState(() {
              _state = _DeployStateKind.smokeFail;
              _errorTitle = 'Smoke test failed';
              _errorBody = error.message;
            });
            return;
          case _InappFail(:final error):
            setState(() {
              _state = _DeployStateKind.inappFail;
              _errorTitle = "Couldn't start in-app channel";
              _errorBody = error.message;
            });
            return;
          case _Cancelled():
            setState(() => _state = _DeployStateKind.editing);
            return;
        }
      }
      // build() composes WizardShell(currentStep: 2) with: name TextField +
      // (conditional) telegram toggle row + (conditional) ChannelInputs +
      // (state-driven) Deploy button OR loading card OR fail card.
    }
    ```
