    @riverpod
    class ChatScope extends _$ChatScope {
      late MessagesStream _stream;
      CancelToken? _historyToken;
      StreamSubscription<SseEvent>? _sseSub;
      ProviderSubscription<AppLifecycleState>? _lifeSub;

      @override
      ChatState build(String agentInstanceId) {
        // Initialize stream + subscriptions; ref.onDispose cleans up.
        _stream = MessagesStream(
          baseUrl: ref.read(appEnvProvider).baseUrl,
          agentId: agentInstanceId,
          cookieProvider: () => ref.read(secureStorageProvider).readSessionId(),
        );
        _bootstrap();
        _lifeSub = ref.listen(appLifecycleProvider, _onLifecycleChanged, fireImmediately: false);

        ref.onDispose(() {
          _historyToken?.cancel();
          _sseSub?.cancel();
          _stream.dispose();
          _lifeSub?.close();
        });
        return const ChatState();
      }

      Future<void> _bootstrap() async {
        // D-36 — parallel GET history + SSE connect.
        _historyToken = CancelToken();
        final historyFuture = ref.read(apiClientProvider).messagesHistory(
          agentId: agentInstanceId,
          limit: 200,
          cancelToken: _historyToken,
        );
        _sseSub = _stream.events.listen(_onSseEvent);
        await _stream.connect();
        final historyRes = await historyFuture;
        if (historyRes case Ok(:final value)) {
          state = state.upsertMany(value.messages);
        }
      }

      void _onLifecycleChanged(AppLifecycleState? prev, AppLifecycleState next) async {
        if (prev != next && next == AppLifecycleState.resumed) {
          await _stream.disconnect();
          await _stream.connect();   // wrapper re-injects Last-Event-Id
        }
      }

      void _onSseEvent(SseEvent ev) {
        // READ spikes/flutter-sse-envelope-inspect.md FIRST.
        // The Wave 0 Spike B artifact has a `dedup_key:` line with one of two verdicts.
        //
        // Dedup-key branching (per RESEARCH §Open Questions Q1 RESOLVED):
        //   If envelope payload has inapp_message_id:
        //     final dedupKey = msg.inappMessageId;
        //   Else (only seq present):
        //     final dedupKey = msg.seq.toString();
        //
        // Concrete shape:
        //   final json = jsonDecode(ev.data) as Map<String, dynamic>;
        //   final msg = ChatMessage.fromJson(json);
        //   final dedupKey = msg.inappMessageId ?? msg.seq.toString();
        //   state = state.upsertOne(dedupKey, msg);
        //
        // The seq branch additionally cross-correlates with history rows on
        // first arrival (match by content+role+createdAt within ±2s) to avoid
        // duplicate bubbles for the same logical message.
      }

      Future<void> sendMessage(String content) async {
        final idemKey = const Uuid().v4();
        // 1. Optimistic insert (D-41).
        state = state.insertOptimistic(idemKey: idemKey, content: content);
        // 2. POST with cancelable.
        final r = await ref.read(apiClientProvider).postMessage(
          agentId: agentInstanceId,
          content: content,
          idempotencyKey: idemKey,
        );
        switch (r) {
          case Ok(): /* SSE will deliver assistant + user-mirror; dedup map handles. */
          case Err(:final error): state = state.markFailed(idemKey: idemKey, error: error.message);
        }
      }

      Future<void> retryFailed({required String failedId, required String content}) async {
        // D-45 — generate NEW Uuid().v4(); failed bubble stays in history.
        final newKey = const Uuid().v4();
        state = state.insertOptimistic(idemKey: newKey, content: content);
        final r = await ref.read(apiClientProvider).postMessage(
          agentId: agentInstanceId, content: content, idempotencyKey: newKey,
        );
        if (r is Err<MessagePostAck>) {
          state = state.markFailed(idemKey: newKey, error: r.error.message);
        }
      }
    }

    class AssistantBubble extends StatelessWidget {
      const AssistantBubble({required this.content, this.onCopy, this.onSelect, super.key});
      final String content;
      final VoidCallback? onCopy;
      final VoidCallback? onSelect;

      static Future<void> _onTapLink(String text, String? href, String title) async {
        if (href == null) return;
        final uri = Uri.tryParse(href);
        if (uri == null) return;

    // Phase 25 Wave 4 D-47 — inline grouped timestamp divider.
    // Rendered between bubbles ONLY when next.createdAt - prev.createdAt > 5 minutes.
    // Caption format:
    //   same calendar day  → 'HH:mm' (24h, e.g. '14:32')
    //   different day      → 'MMM dd HH:mm' (e.g. 'Apr 28 14:32')
    // Color: SolvrColors.mutedForeground; size: 12; centered.
    //
    // Per D-47: NO timestamp on every bubble; NO long-press-to-reveal.

    import 'package:agent_playground/core/api/dtos.dart';
    import 'package:agent_playground/core/theme/solvr_theme.dart';
    import 'package:flutter/material.dart';
    import 'package:intl/intl.dart';

    class TimestampDivider extends StatelessWidget {
      const TimestampDivider({required this.label, super.key});
      final String label;

      static const Duration _gapThreshold = Duration(minutes: 5);

      /// Returns a TimestampDivider widget if a divider is warranted between
      /// `prev` and `next`, otherwise null. Caller injects null whenever the
      /// returned widget is null (or wraps it in a conditional spread).
      static Widget? maybeBuild({required ChatMessage? prev, required ChatMessage next}) {
        if (prev == null) return null;            // no divider above first bubble
        final p = DateTime.tryParse(prev.createdAt ?? '');
        final n = DateTime.tryParse(next.createdAt ?? '');
        if (p == null || n == null) return null;  // graceful degradation
        if (n.difference(p) <= _gapThreshold) return null;
        final now = DateTime.now();
        final sameDay = n.year == now.year && n.month == now.month && n.day == now.day;
        final fmt = sameDay ? DateFormat('HH:mm') : DateFormat('MMM dd HH:mm');
        return TimestampDivider(label: fmt.format(n.toLocal()));
      }

      @override
      Widget build(BuildContext context) => Padding(
        padding: const EdgeInsets.symmetric(vertical: 12, horizontal: 16),
        child: Center(
          child: Text(
            label,
            style: const TextStyle(
              fontSize: 12,
              color: SolvrColors.mutedForeground,
            ),
          ),
        ),
      );
    }
