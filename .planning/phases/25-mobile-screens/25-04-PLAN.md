    ```dart
    // Phase 25 Wave 2 — Dashboard providers.
    // - agentsListProvider: re-fetched on mount + AppLifecycleState.resumed
    //   + manual pull-to-refresh. CancelToken concurrency guard (Pitfall #8).
    // - recipesProvider: cached AsyncValue; drives AsciiAgentBanner names stream
    //   in the empty-state path (D-17). Refetched per wizard open in Wave 3,
    //   but Dashboard reads it for the empty-state banner only.
    // - recipeNamesStreamProvider: Stream<List<String>> derived from recipesProvider
    //   so AsciiAgentBanner stays a pure widget (UI-SPEC line 524).

    import 'dart:async';

    import 'package:agent_playground/core/api/dtos.dart';
    import 'package:agent_playground/core/api/providers.dart';
    import 'package:agent_playground/core/api/result.dart';
    import 'package:agent_playground/core/lifecycle/app_lifecycle_observer.dart';
    import 'package:dio/dio.dart';
    import 'package:flutter/widgets.dart';
    import 'package:flutter_riverpod/flutter_riverpod.dart';
    import 'package:riverpod_annotation/riverpod_annotation.dart';

    part 'dashboard_providers.g.dart';

    @riverpod
    class AgentsList extends _$AgentsList {
      CancelToken? _cancel;

      @override
      Future<List<AgentSummary>> build() async {
        _cancel?.cancel();
        final cancel = _cancel = CancelToken();
        ref.onDispose(() => cancel.cancel());

        // D-12 — refetch on AppLifecycleState.resumed.
        ref.listen(appLifecycleProvider, (prev, next) {
          if (prev != next && next == AppLifecycleState.resumed) {
            ref.invalidateSelf();
          }
        });

        final api = ref.watch(apiClientProvider);
        final r = await api.agentsList(cancelToken: cancel);
        return switch (r) {
          Ok(:final value) => _sortByLastActivityDesc(value),
          Err(:final error) => throw error,
        };
      }

      List<AgentSummary> _sortByLastActivityDesc(List<AgentSummary> list) {
        // D-13 — most recent first; null lastActivity falls to bottom.
        final sorted = [...list]..sort((a, b) {
          final aa = a.lastActivity ?? '';
          final bb = b.lastActivity ?? '';
          return bb.compareTo(aa);
        });
        return sorted;
      }
    }

    @Riverpod(keepAlive: true)
    Future<List<Recipe>> recipes(Ref ref) async {
      final api = ref.watch(apiClientProvider);
      final r = await api.recipes();
      return switch (r) {
        Ok(:final value) => value,
        Err(:final error) => throw error,
      };
    }

    /// AsciiAgentBanner expects Stream<List<String>>; recipes is a one-shot
    /// AsyncValue. Wrap in a single-emit stream so the widget stays pure.
    @riverpod
    Stream<List<String>> recipeNamesStream(Ref ref) async* {
      final list = await ref.watch(recipesProvider.future);
      yield list.map((r) => r.name).toList(growable: false);
      // No subsequent emissions — recipesProvider invalidation re-runs build.
    }
    ```

    ```dart
    class AgentRow extends StatelessWidget {
      const AgentRow({required this.agent, required this.onTap, super.key});

      final AgentSummary agent;
      final VoidCallback onTap;

      @override
      Widget build(BuildContext context) {
        final caption = Theme.of(context).textTheme.bodySmall;
        final body = Theme.of(context).textTheme.bodyLarge;
        return InkWell(
          onTap: onTap,
          child: Padding(
            padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 8),
            child: SizedBox(
              height: 56,
              child: Row(
                children: [
                  StatusDot(status: agent.status),
                  const SizedBox(width: 12),
                  Expanded(
                    child: Column(
                      crossAxisAlignment: CrossAxisAlignment.start,
                      mainAxisAlignment: MainAxisAlignment.center,
                      children: [
                        Text(
                          agent.name,
                          maxLines: 1,
                          overflow: TextOverflow.ellipsis,
                          style: body?.copyWith(fontWeight: FontWeight.w600),
                        ),
                        Text(
                          agent.model,
                          maxLines: 1,
                          overflow: TextOverflow.ellipsis,
                          style: SolvrTextStyles.mono(fontSize: 12).copyWith(
                            color: SolvrColors.mutedForeground,
                          ),
                        ),
                      ],
                    ),
                  ),
                  const SizedBox(width: 8),
                  Text(
                    _relativeTime(agent.lastActivity),
                    style: caption?.copyWith(color: SolvrColors.mutedForeground),
                  ),
                ],
              ),
            ),
          ),
        );
      }

      static String _relativeTime(String? iso) {
        if (iso == null || iso.isEmpty) return '';
        final ts = DateTime.tryParse(iso);
        if (ts == null) return '';
        final delta = DateTime.now().difference(ts);
        if (delta.inMinutes < 1) return 'now';
        if (delta.inMinutes < 60) return '${delta.inMinutes}m';
        if (delta.inHours < 24) return '${delta.inHours}h';
        if (delta.inDays < 2) return 'yesterday';
        if (delta.inDays < 7) return '${delta.inDays}d';
        // Older: ISO short — keep simple, no localization in MVP.
        return ts.toIso8601String().substring(0, 10);
      }
    }
    ```

    ```dart
    class DashboardScreen extends ConsumerWidget {

      Widget build(BuildContext context, WidgetRef ref) {

        Widget body;
        if (agents.isLoading) {

            body = _EmptyState(namesStream: namesStream, context: context);
          } else {
            body = _PopulatedState(
              agents: list,
              namesStream: namesStream,

              onRefresh: () async => ref.invalidate(agentsListProvider),
            );
          }
        }

          appBar: AppBar(
            title: const Text('>_ SOLVR_LABS'),
            actions: [
              PopupMenuButton<String>(
                onSelected: (v) async {
                  if (v == 'signout') {
                    final result = await ConfirmDialog.show(
                      context,
                      title: 'Sign out of Solvr Labs?',
                      cancelLabel: 'Cancel',
                      confirmLabel: 'Sign out',
    ```
