    """ProxyIPMap — in-process {bridge_ip -> (user_id, agent_instance_id, inapp_auth_token)}
    cache. Source: agent_containers.bridge_ip column (AMD-04, migration 013).
    Refresh task polls every 60s (D-07; PROBE-VAL-10 confirmed events() not needed).
    """
    from __future__ import annotations

    import asyncio
    import logging
    from typing import Any
    from uuid import UUID

    _log = logging.getLogger("api_server.proxy_ip_map")

    PROXY_IP_REFRESH_INTERVAL_S: float = 60.0


    class ProxyIPMap:
        def __init__(self, db_pool: Any) -> None:
            self._db_pool = db_pool
            self._lock = asyncio.Lock()
            self._cache: dict[str, tuple[UUID, UUID, str]] = {}

        async def refresh(self) -> int:
            async with self._db_pool.acquire() as conn:
                rows = await conn.fetch("""
                    SELECT bridge_ip, user_id, agent_instance_id, inapp_auth_token
                    FROM agent_containers
                    WHERE container_status = 'running'
                      AND bridge_ip IS NOT NULL
                      AND inapp_auth_token IS NOT NULL
                """)
            new_cache: dict[str, tuple[UUID, UUID, str]] = {
                str(r["bridge_ip"]): (r["user_id"], r["agent_instance_id"], r["inapp_auth_token"])
                for r in rows
            }
            async with self._lock:
                self._cache = new_cache
            return len(new_cache)

        async def get(self, bridge_ip: str) -> tuple[UUID, UUID, str] | None:
            async with self._lock:
                return self._cache.get(bridge_ip)


    async def refresh_loop(
        ip_map: ProxyIPMap,
        stop_event: asyncio.Event,
        interval_s: float = PROXY_IP_REFRESH_INTERVAL_S,
    ) -> None:
        while not stop_event.is_set():
            try:
                await ip_map.refresh()
            except asyncio.CancelledError:
                raise
            except Exception:
                _log.exception("proxy_ip_map.refresh_failed")
            try:
                await asyncio.wait_for(stop_event.wait(), timeout=interval_s)
                return
            except asyncio.TimeoutError:
                pass
    ```

    async def insert_reserved_row(
        conn: asyncpg.Connection,
        *,
        user_id: UUID,
        key: str,
        request_body_hash: str,
        ttl_seconds: int = 86400,
    ) -> bool:
        """Insert an in_flight reserved row. Returns True if WE inserted it, False if it already existed.
        Caller treats False as "another caller is in-flight; poll for completion."
        AMD-03 — closes the in-flight gap surfaced by PROBE-VAL-06.
        """
        result = await conn.execute(
            """
            INSERT INTO idempotency_keys (
                user_id, key, request_body_hash, verdict_json, status, expires_at, created_at
            ) VALUES ($1, $2, $3, NULL, 'in_flight', NOW() + ($4 || ' seconds')::interval, NOW())
            ON CONFLICT (user_id, key) DO NOTHING
            """,
            user_id, key, request_body_hash, str(ttl_seconds),
        )
        # asyncpg returns 'INSERT 0 1' on insert, 'INSERT 0 0' on conflict-skip
        return result.endswith(" 1")


    async def poll_for_completion(
        conn: asyncpg.Connection,
        *,
        user_id: UUID,
        key: str,
        max_wait_s: float = 5.0,
        poll_interval_s: float = 0.1,
    ) -> dict | None:
        """Wait for a reserved row to flip to status='success'. Returns the verdict_json
        on success, None on timeout or status='failed'.
        """
        deadline = asyncio.get_event_loop().time() + max_wait_s
        while True:
            row = await conn.fetchrow(
                "SELECT status, verdict_json FROM idempotency_keys WHERE user_id = $1 AND key = $2",
                user_id, key,
            )
            if row is None:
                return None
            if row["status"] == "success":
                return row["verdict_json"]  # already a dict if JSONB
            if row["status"] == "failed":
                return None
            if asyncio.get_event_loop().time() >= deadline:
                return None
            await asyncio.sleep(poll_interval_s)


    async def finalize_reserved_row(
        conn: asyncpg.Connection,
        *,
        user_id: UUID,
        key: str,
        verdict_json: dict,
        status: str = "success",
    ) -> None:
        """Update an in_flight row to its terminal status + verdict."""
        if status not in ("success", "failed"):
            raise ValueError(f"finalize status must be 'success' or 'failed', got {status!r}")
        await conn.execute(
            """
            UPDATE idempotency_keys
            SET status = $3, verdict_json = $4
            WHERE user_id = $1 AND key = $2
            """,
            user_id, key, status, json.dumps(verdict_json) if not isinstance(verdict_json, str) else verdict_json,
        )
    ```

    @router.post("/llm/forward/{path:path}")
    async def forward(path: str, request: Request) -> StreamingResponse:
        # 1. Identify caller via IP-map + Authorization Bearer match (D-07 defense-in-depth)
        bridge_ip = request.client.host
        caller = await request.app.state.proxy_ip_map.get(bridge_ip)
        if caller is None:
            return JSONResponse(make_error_envelope("unauthorized", "unknown caller"), status_code=401)
        user_id, agent_instance_id, expected_token = caller

        provider, key = await request.app.state.proxy_byok_cache.get(user_id, agent_instance_id)
        if key is None:

        spec = PROVIDERS[provider]

        # 3. Read + mutate body (D-08 + D-14)
        body_bytes = await request.body()

        if provider == "openrouter":

        body_out = json.dumps(body, separators=(",", ":")).encode()
        body_hash = hashlib.sha256(body_out).hexdigest()

        # 4. Build upstream URL + headers
        upstream_url = f"{spec.base_url}/{path}"
        headers = {
            "Content-Type": "application/json",
            "Content-Length": str(len(body_out)),
            spec.auth_header_name: spec.auth_value_template.format(key=key),
            **spec.extra_headers,
        }
        if provider == "anthropic":
            headers["OpenTelemetry"] = f"ap_user={user_id}"

        # 5. Idempotency (D-16 + AMD-03 reserved-row)
        idem_key = request.headers.get("Idempotency-Key")
        if idem_key:

                )

                    if cached is not None:

                            )
                except Exception as exc:

            _gen(),
            status_code=status_code,
            media_type=upstream_resp.headers.get("content-type", "application/json"),
