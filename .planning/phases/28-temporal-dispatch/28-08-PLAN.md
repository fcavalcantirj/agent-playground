import 'package:flutter/material.dart';

void main() {
  testWidgets('UsageTickerWidget survives Dashboard → Chat → Dashboard navigation', (tester) async {
    // Override usageSummaryProvider to a stable AsyncValue.data so the ticker
    // renders without making real HTTP calls.
    final overrides = [
      usageSummaryProvider.overrideWith((ref) async {
        // Build a minimal UsageSummary the widget renders. Inspect
        // usage_models.dart for the constructor.
        return /* a UsageSummary instance with byAgent: [] */;
      }),
    ];

    await tester.pumpWidget(
      ProviderScope(
        overrides: overrides,
        child: MaterialApp(
          home: const _DashboardStub(),
        ),
      ),
    );
    await tester.pumpAndSettle();

    // Verify the ticker is visible
    expect(find.byType(UsageTickerWidget), findsOneWidget);

    // Tap into Chat (use the stub's "Open Chat" button)
    await tester.tap(find.text('Open Chat'));
    await tester.pumpAndSettle();
    expect(find.byType(UsageTickerWidget), findsOneWidget);   // visible in chat too

    // Pop back to Dashboard
    final BuildContext ctx = tester.element(find.byType(UsageTickerWidget));
    Navigator.of(ctx).pop();
    await tester.pumpAndSettle();
    expect(find.byType(UsageTickerWidget), findsOneWidget);  // back on dashboard

    // CRITICAL — no defunct-element exception fired during pump cycles.
    // The test framework surfaces such exceptions via tester.takeException().
    expect(tester.takeException(), isNull);
  });
}
