```
#!/usr/bin/env python3
"""Phase 28 cutover — one-shot sweep of legacy stuck inapp_messages rows.

Run BEFORE rolling api_server to the Phase 28 build. Rows that the legacy
`inapp_dispatcher` left in `status='forwarded'` past the 11-minute reaper
threshold are orphaned by the cutover (no in-flight workflow to claim them).

This script transitions them to `status='failed'` with
`last_error='phase28_cutover_swept'` so the audit log is preserved and
the user surface shows a terminal failure instead of a perpetual "typing".

Usage:
    DATABASE_URL=postgresql://... python tools/migrate_phase28_stuck_rows.py
    DATABASE_URL=postgresql://... python tools/migrate_phase28_stuck_rows.py --dry-run

Idempotent: re-running after the sweep is a no-op (zero rows match WHERE).
"""
from __future__ import annotations

import argparse
import asyncio
import logging
import os
import sys

import asyncpg

log = logging.getLogger("phase28.cutover")

SWEEP_SQL = """
UPDATE inapp_messages
SET status = 'failed',
    last_error = 'phase28_cutover_swept',
    completed_at = NOW()
WHERE status = 'forwarded'
  AND last_attempt_at < NOW() - INTERVAL '11 minutes'
RETURNING id, user_id, agent_id, last_attempt_at
"""

DRY_RUN_SQL = """
SELECT id, user_id, agent_id, last_attempt_at
FROM inapp_messages
WHERE status = 'forwarded'
  AND last_attempt_at < NOW() - INTERVAL '11 minutes'
"""


async def main() -> int:
    parser = argparse.ArgumentParser(description="Phase 28 cutover sweep")
    parser.add_argument("--dry-run", action="store_true", help="show what would change, do not write")
    parser.add_argument("--limit", type=int, default=10_000, help="safety cap; abort if more rows match")
    args = parser.parse_args()

    logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s")

    dsn = os.environ.get("DATABASE_URL")
    if not dsn:
        log.error("DATABASE_URL env var is required")
        return 2

    # asyncpg DSN doesn't accept the "+asyncpg" driver suffix
    dsn_clean = dsn.replace("postgresql+asyncpg://", "postgresql://")

    conn = await asyncpg.connect(dsn_clean)
    try:
        if args.dry_run:
            rows = await conn.fetch(DRY_RUN_SQL)
            log.info("phase28.cutover.dry_run", extra={"row_count": len(rows)})
            for r in rows[:50]:
                log.info("phase28.cutover.row", extra={"id": str(r["id"]), "user_id": str(r["user_id"]), "agent_id": str(r["agent_id"]), "last_attempt_at": r["last_attempt_at"].isoformat()})
            if len(rows) > 50:
                log.info("phase28.cutover.dry_run.truncated", extra={"shown": 50, "total": len(rows)})
            return 0
        async with conn.transaction():
            # Pre-count for safety
            count = await conn.fetchval("SELECT COUNT(*) FROM inapp_messages WHERE status='forwarded' AND last_attempt_at < NOW() - INTERVAL '11 minutes'")
            if count > args.limit:
                log.error("phase28.cutover.safety_cap_exceeded", extra={"count": count, "limit": args.limit})
                return 3
            rows = await conn.fetch(SWEEP_SQL)
            log.info("phase28.cutover.swept", extra={"row_count": len(rows)})
            for r in rows:
                log.info("phase28.cutover.swept.row", extra={"id": str(r["id"]), "user_id": str(r["user_id"]), "agent_id": str(r["agent_id"]), "last_attempt_at": r["last_attempt_at"].isoformat()})
        return 0
    finally:
        await conn.close()


if __name__ == "__main__":
    sys.exit(asyncio.run(main()))
```
